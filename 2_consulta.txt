Cómo se referencia una clase desde un procedimiento de ventana winapi c++

La forma correcta de referenciar una clase C++ desde un procedimiento de ventana WinAPI (WNDPROC) depende de cómo quieras vincular la instancia de tu clase con la ventana. WinAPI es C puro, así que no entiende clases ni métodos, pero sí permite guardar punteros dentro de la ventana para recuperarlos después.

La técnica estándar y recomendada es usar GWLP_USERDATA o WM_NCCREATE + CREATESTRUCT.

🎯 Resumen rápido
La manera correcta es:

Pasar el puntero a tu clase cuando creas la ventana (CreateWindowEx).

Guardarlo en la ventana durante WM_NCCREATE.

Recuperarlo en cada mensaje dentro del WNDPROC.


1. Tu clase C++
cpp
class MiVentana {
public:
    void OnClick() {
        MessageBox(NULL, L"Click!", L"Clase", MB_OK);
    }
};


2. Crear la ventana pasando el puntero

MiVentana* obj = new MiVentana();

HWND hwnd = CreateWindowEx(
    0,
    L"MiClaseWinAPI",
    L"Demo",
    WS_OVERLAPPEDWINDOW,
    CW_USEDEFAULT, CW_USEDEFAULT,
    400, 300,
    NULL,
    NULL,
    hInstance,
    obj   // <--- aquí pasas el puntero a la clase
);


3. Guardar el puntero en WM_NCCREATE
cpp
LRESULT CALLBACK WndProc(HWND hwnd, UINT msg, WPARAM wParam, LPARAM lParam)
{
    if (msg == WM_NCCREATE) {
        CREATESTRUCT* cs = reinterpret_cast<CREATESTRUCT*>(lParam);
        MiVentana* obj = reinterpret_cast<MiVentana*>(cs->lpCreateParams);

        SetWindowLongPtr(hwnd, GWLP_USERDATA, reinterpret_cast<LONG_PTR>(obj));
        return TRUE;
    }
4. Recuperarlo en cada mensaje

    MiVentana* obj = reinterpret_cast<MiVentana*>(
        GetWindowLongPtr(hwnd, GWLP_USERDATA)
    );

    switch (msg) {
    case WM_LBUTTONDOWN:
        if (obj) obj->OnClick();
        return 0;

    case WM_DESTROY:
        PostQuitMessage(0);
        return 0;
    }

    return DefWindowProc(hwnd, msg, wParam, lParam);
}


¿Por qué funciona esto?
WinAPI permite asociar datos arbitrarios a una ventana mediante:

lpCreateParams en CREATESTRUCT (recibido en WM_NCCREATE)

GWLP_USERDATA para almacenar un puntero de 64 bits

Esto está documentado en la sección “Asociating a Window Procedure with a Window Class” de Microsoft Learn .

🛑 Errores comunes
❌ Intentar llamar métodos de clase directamente desde el WNDPROC
→ WinAPI no sabe qué es this.

❌ Guardar el puntero en WM_CREATE  
→ Es demasiado tarde; la ventana ya está creada.

❌ Usar variables globales
→ Funciona, pero rompe el diseño orientado a objetos.

La forma correcta de asociar clases C++ (objetos) con un procedimiento de ventana WinProc en WinAPI es usar los datos extra de la ventana (GWLP_USERDATA) o los extra bytes de clase/ventana (cbWndExtra, cbClsExtra) para almacenar un puntero a la instancia de tu clase, y luego recuperarlo dentro del WndProc.  Esto está documentado en WinAPI como “almacenar datos de usuario con una ventana” 

🎯 Resumen rápido
La técnica estándar es:

1.- Crear la ventana.

2.- Guardar el puntero a tu clase en la ventana usando:

	SetWindowLongPtr(hwnd, GWLP_USERDATA, (LONG_PTR)this);

3.- Recuperarlo dentro del WinProc:

	MyClass* pThis = (MyClass*)GetWindowLongPtr(hwnd, GWLP_USERDATA);

4.- Usar pThis para llamar métodos de tu clase desde el WinProc.

¿Por qué es necesario?
WinAPI exige que el procedimiento de ventana sea una función global o estática:

	LRESULT CALLBACK WndProc(HWND, UINT, WPARAM, LPARAM);

Pero tú quieres que cada ventana tenga su propio objeto C++ con estado.
La solución es asociar el HWND con tu objeto.

Método recomendado (GWLP_USERDATA)
1. Clase C++

class MiVentana {
public:
    HWND hwnd;

    void Crear(HINSTANCE hInst) {
        hwnd = CreateWindowEx(
            0, L"MiClaseWin", L"Titulo",
            WS_OVERLAPPEDWINDOW,
            CW_USEDEFAULT, CW_USEDEFAULT,
            400, 300,
            nullptr, nullptr, hInst, this   // <--- pasar this
        );
    }

    LRESULT ProcesarMensaje(UINT msg, WPARAM wp, LPARAM lp) {
        switch(msg) {
            case WM_PAINT:
                // lógica de la clase
                return 0;
        }
        return DefWindowProc(hwnd, msg, wp, lp);
    }
};

2. WinProc global

LRESULT CALLBACK WndProc(HWND hwnd, UINT msg, WPARAM wp, LPARAM lp)
{
    MiVentana* pThis;

    if (msg == WM_NCCREATE) {
        // Extraer el puntero pasado en CreateWindowEx
        CREATESTRUCT* cs = (CREATESTRUCT*)lp;
        pThis = (MiVentana*)cs->lpCreateParams;

        // Guardarlo en la ventana
        SetWindowLongPtr(hwnd, GWLP_USERDATA, (LONG_PTR)pThis);

        // Guardar el HWND dentro del objeto
        pThis->hwnd = hwnd;
    }

    // Recuperar el puntero
    pThis = (MiVentana*)GetWindowLongPtr(hwnd, GWLP_USERDATA);

    if (pThis) {
        return pThis->ProcesarMensaje(msg, wp, lp);
    }

    return DefWindowProc(hwnd, msg, wp, lp);
}


Alternativa: usar SetProp (menos común)
WinAPI también permite asociar datos arbitrarios con una ventana:

	SetProp(hwnd, TEXT("MiData"), (HANDLE)pThis);
	MiVentana* pThis = (MiVentana*)GetProp(hwnd, TEXT("MiData"));


Alternativa moderna: SetWindowSubclass
Si estás subclasando controles comunes (botones, edit, etc.):

	SetWindowSubclass(hwnd, MiWndProc, 0, (DWORD_PTR)this);

Y luego recuperas el puntero en el WinProc de la subclase:

	MiClase* pThis = (MiClase*)dwRefData;


¿Qué método debes usar?

| Caso 					     | Método recomendado 			|
| --- 					     | --- 					|
| Ventanas creadas por tu aplicación         | ``GWLP_USERDATA`` + ``WM_NCCREATE`` 	|
| Controles estándar (Button, Edit…)         | ``SetWindowSubclass`` 			|
| Necesitas asociar varios datos arbitrarios | ``SetProp`` 				|

int WINAPI wWinMain(HINSTANCE hInst, HINSTANCE, PWSTR, int nCmdShow)
{
    WNDCLASS wc = { };
    wc.lpfnWndProc   = WndProc;
    wc.hInstance     = hInst;
    wc.lpszClassName = L"MiClaseWin";

    RegisterClass(&wc);

    MiVentana ventana;
    ventana.Crear(hInst);

    ShowWindow(ventana.hwnd, nCmdShow);

    MSG msg;
    while (GetMessage(&msg, nullptr, 0, 0)) {
        TranslateMessage(&msg);
        DispatchMessage(&msg);
    }

    return 0;
}

Aquí está el problema **exactamente identificado** y la solución **precisa** para que tu método `setBaseNueva()` sea reconocido dentro del `WndProc`.

---

# 🎯 **El error real en tu código**
En tu `WndProc` tienes esta línea:

```cpp
void G1.setBaseNueva(100, 12);
```

Esto es **inválido en C++** por dos razones:

1. **No se puede poner `void` delante de una llamada a método.**  
2. La sintaxis correcta para llamar un método es simplemente:

```cpp
G1.setBaseNueva(100, 12);
```

---

# ✔️ **Corrección mínima necesaria**
Cambia:

```cpp
void G1.setBaseNueva(100, 12);
```

Por:

```cpp
G1.setBaseNueva(100, 12);
```

---

# 🧠 **Pero hay otro problema más importante**
Tu objeto:

```cpp
Generador G1(150, 20, 1, 2);
```

Está **dentro del WndProc**, lo que significa:

- Se crea **cada vez que llega un mensaje**.
- Se destruye **al salir del WndProc**.
- No conserva estado.
- No sirve para cálculos persistentes.
- No podrás usarlo en otros mensajes (WM_COMMAND, WM_PAINT, etc.).

Por eso tus getters funcionan (los llamas inmediatamente),  
pero `setBaseNueva()` parece “no funcionar” porque el objeto **desaparece**.

---

# 🛠️ **Solución correcta: hacer el objeto persistente**
### Opción 1 — Declararlo `static` dentro del WndProc

```cpp
static Generador G1(150, 20, 1, 2);
```

Y luego:

```cpp
G1.setBaseNueva(100, 12);
```

Esto hace que el objeto viva **toda la vida de la ventana**.

---

### Opción 2 — Guardarlo en `GWLP_USERDATA` (más profesional)

Si quieres que cada ventana tenga su propio objeto:

```cpp
case WM_NCCREATE:
{
    CREATESTRUCT* cs = (CREATESTRUCT*)lParam;
    Generador* gen = new Generador(150, 20, 1, 2);
    SetWindowLongPtr(hWnd, GWLP_USERDATA, (LONG_PTR)gen);

break;
```

Luego lo recuperas:

```cpp
Generador* G1 = (Generador*)GetWindowLongPtr(hWnd, GWLP_USERDATA);
G1->setBaseNueva(100, 12);
```

Y al destruir la ventana:

```cpp

{

break;
```

LRESULT CALLBACK WndProc(HWND hWnd, UINT message, WPARAM wParam, LPARAM lParam)
{
    Generador* G1 = (Generador*)GetWindowLongPtr(hWnd, GWLP_USERDATA);

    switch (message)
    {
    case WM_NCCREATE:
    {
        CREATESTRUCT* cs = (CREATESTRUCT*)lParam;

        Generador* gen = new Generador(150, 20, 1, 2);
        SetWindowLongPtr(hWnd, GWLP_USERDATA, (LONG_PTR)gen);

        return TRUE;
    }

    case WM_COMMAND:
    {
        int wmId = LOWORD(wParam);

        switch (wmId)
        {
        case IDM_NUEVO:
            if (G1)
            {
                G1->setBaseNueva(100, 12);

                double P = G1->getMVA();
                double XdNuevo = G1->getXdN();

                wchar_t texto[64];
                swprintf_s(texto, 64, L"MVA=%.2f   Xd nueva=%.3f", P, XdNuevo);

                MessageBoxW(hWnd, texto, L"RESULTADO", MB_OK);
            }
            break;
        }
    }
    break;

    case WM_DESTROY:
    {
        Generador* G1 = (Generador*)GetWindowLongPtr(hWnd, GWLP_USERDATA);
        delete G1;
        PostQuitMessage(0);
    }
    break;
    }

    return DefWindowProc(hWnd, message, wParam, lParam);
}
